SELECT year, SUM(CASE WHEN party_major='D' THEN candidate_votes ELSE 0 END) d, SUM(CASE WHEN party_major='R' THEN candidate_votes ELSE 0 END) r, SUM(candidate_votes) tot, ROUND(100.0*SUM(CASE WHEN party_major='D' THEN candidate_votes ELSE 0 END)/SUM(candidate_votes),1) dpct, ROUND(100.0*SUM(CASE WHEN party_major='R' THEN candidate_votes ELSE 0 END)/SUM(candidate_votes),1) rpct, ROUND(100.0*(SUM(CASE WHEN party_major='D' THEN candidate_votes ELSE 0 END)-SUM(CASE WHEN party_major='R' THEN candidate_votes ELSE 0 END))/SUM(candidate_votes),1) margin FROM officials.house_election_results WHERE stage='gen' AND special=false AND (runoff=false OR runoff IS NULL) AND (uncontested=false OR uncontested IS NULL) AND (writein=false OR writein IS NULL) AND candidate_kind='candidate' GROUP BY year ORDER BY year
