WITH s AS (SELECT congress, state_name, bioguide_id, CASE WHEN party_name LIKE 'Democ%' OR party_name LIKE 'Independent%' AND bioguide_id IN ('S000033','K000383') THEN 'D' WHEN party_name LIKE 'Republican%' THEN 'R' ELSE party_name END p FROM officials.members WHERE district IS NULL), d AS (SELECT congress, state_name, COUNT(DISTINCT p) np, COUNT(*) n, MIN(p) mn, MAX(p) mx FROM s GROUP BY congress, state_name) SELECT congress, COUNT(*) states, SUM(CASE WHEN np>1 THEN 1 ELSE 0 END) split_loose, SUM(CASE WHEN n=2 AND np=2 THEN 1 ELSE 0 END) split_exact2, SUM(CASE WHEN n=2 AND np=1 AND mn='D' THEN 1 ELSE 0 END) uD, SUM(CASE WHEN n=2 AND np=1 AND mn='R' THEN 1 ELSE 0 END) uR FROM d GROUP BY congress ORDER BY congress
